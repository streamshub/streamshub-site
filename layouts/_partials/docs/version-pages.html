{{/* Given a docs-root page (e.g. /console/docs/), returns its version pages sorted: development branch first, then tags numerically descending. Each entry is a dict with: - Page: the Hugo page object - Version: the raw folder name (e.g. "main", "0.13.0") - Display: user-facing label ("In-Development" for dev branches, raw version for tags) - Key: sort key (dev branch = "9", tags = zero-padded semver) Development branches are detected by the developmentBranch front matter flag (stamped by docBuilder). As a backwards-compatibility fallback, any non-semver folder without the flag explicitly set to false is also treated as a development branch — this covers docs built before the flag was introduced. */}} {{- $versionPages := slice -}} {{- range .Pages -}} {{- $version := path.Base .Path -}} {{- $isDev := .Params.developmentBranch -}} {{- $isTag := findRE `^[0-9]+(\.[0-9]+){1,2}$` $version -}} {{- $isLegacyDev := and (not $isTag) (not (isset .Params "developmentbranch")) -}} {{- if or $isDev $isTag $isLegacyDev -}} {{- $display := $version -}} {{- $key := "" -}} {{- if or $isDev $isLegacyDev -}} {{- $display = "In-Development" -}} {{- $key = "9" -}} {{- else -}} {{- $padded := slice -}} {{- range split $version "." -}} {{- $padded = $padded | append (printf "%05d" (int .)) -}} {{- end -}} {{- $key = printf "1.%s" (delimit $padded ".") -}} {{- end -}} {{- $versionPages = $versionPages | append (dict "Page" . "Version" $version "Display" $display "Key" $key) -}} {{- end -}} {{- end -}} {{- return (sort $versionPages "Key" "desc") -}}
