{{/* Per-project version picker, shown above the section tree on any docs OR quick-start page belonging to one of the three projects. Renders nothing outside those pages (e.g. not on a project's marketing home page or its community page). Version pages are identified the same way as docs/menu-filetree.html: docBuilder tags every version folder's index page with title = the version string (e.g. "main", "0.8.3"). "main" (the development branch) is always the default/top entry; tagged versions sort numerically, newest first. Quick Start lives as a sibling of docs (content/console/quick-start/), not nested inside it - it isn't itself versioned, but the picker should still appear there (defaulting to "main") so readers can jump straight to the full versioned documentation. */}} {{- $versionPattern := `^(main|[0-9]+(\.[0-9]+){1,2})$` -}} {{- $projectRoots := slice (dict "match" "/console/docs" "docs" "/console/docs") (dict "match" "/console/quick-start" "docs" "/console/docs") (dict "match" "/flink-sql-runner/docs" "docs" "/flink-sql-runner/docs") (dict "match" "/strimzi-mcp-server/docs" "docs" "/strimzi-mcp-server/docs") -}} {{- $docsRoot := "" -}} {{- range $projectRoots -}} {{- $docsPath := .docs -}} {{- $matchPage := $.Site.GetPage .match -}} {{- if and $matchPage (or (eq $matchPage $.Page) ($matchPage.IsAncestor $.Page)) -}} {{- $docsRoot = $.Site.GetPage $docsPath -}} {{- end -}} {{- end -}} {{- with $docsRoot -}} {{- $versionPages := slice -}} {{- range .Pages -}} {{- if findRE $versionPattern .Title -}} {{- $versionPages = $versionPages | append . -}} {{- end -}} {{- end -}} {{- if $versionPages -}} {{- $ranked := slice -}} {{- range $versionPages -}} {{- $key := "" -}} {{- if eq .Title "main" -}} {{- $key = "9" -}} {{- else -}} {{- $padded := slice -}} {{- range split .Title "." -}} {{- $padded = $padded | append (printf "%05d" (int .)) -}} {{- end -}} {{- $key = printf "1.%s" (delimit $padded ".") -}} {{- end -}} {{- $ranked = $ranked | append (dict "Page" . "Key" $key) -}} {{- end -}} {{- $sorted := sort $ranked "Key" "desc" -}} {{- $current := $.Page -}} {{- $selected := index $sorted 0 -}} {{- range $sorted -}} {{- if or (eq .Page $current) (.Page.IsAncestor $current) -}} {{- $selected = . -}} {{- end -}} {{- end -}}
